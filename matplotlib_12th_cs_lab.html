                             தரவு காட்சிப்படுத்துதல்
                        

                        

                            தகவல்களையும், தரவுகளையும் வரைகலை அல்லது வரைபட வடிவில் வெளிப்படுத்துவதே தரவு காட்சிப்படுத்துதல் (Data Visualization) எனப்படும். இது சிக்கலான தரவுகளின் போக்கை (trends) எளிதாகக் கண்டறிய உதவுகிறது.
                        

                    

                    

                        

                             மேட்பிளாட்லிப் (Matplotlib)
                        

                        

                            Matplotlib என்பது பைத்தானில் தரவுகளை 2D வரைபடங்களாகக் காட்சிப்படுத்த உதவும் மிகவும் பிரபலமான திறவூற்று (Open Source) நூலகமாகும். இதனை ஜான் டி. ஹண்டர் (John D. Hunter) உருவாக்கினார்.
                        

                    

                    

                        

                             பைபிளாட் (pyplot) இடைமுகம்
                        

                        

                            pyplot என்பது MATLAB போன்ற வரைபட சூழலை வழங்கும் Matplotlib இன் தொகுப்பாகும். இதனை பொதுவாக import matplotlib.pyplot as plt என இறக்குமதி செய்வர்.
                        

                    

                


                
                

                     வரைபடங்களின் வகைகள் (Types of Plots)
                

                

                    

                        
                            

                                வரைபடம் (Chart Type)
                                செயற்கூறு (Function)
                                நோக்கம் மற்றும் பயன்பாடு (Purpose)
                                முக்கிய பண்புக்கூறுகள் (Key Arguments)
                            

                        
                        
                            

                                 கோட்டு வரைபடம் (Line Plot)
                                plt.plot(x, y)
                                தொடர்ச்சியான கால அளவில் ஏற்படும் மாற்றங்களையும் போக்கையும் காட்ட பயன்படுகிறது.
                                color, marker, linewidth, linestyle
                            

                            

                                 பட்டை வரைபடம் (Bar Chart)
                                plt.bar(x, y)
                                வெவ்வேறு பிரிவுகள் அல்லது வகைகளுக்கு இடையேயான அளவுகளை ஒப்பிட பயன்படுகிறது.
                                color, width
                            

                            

                                 வட்ட வரைபடம் (Pie Chart)
                                plt.pie(values, labels=...)
                                மொத்தத்தில் ஒவ்வொரு பகுதியும் எவ்வளவு விகிதம் அல்லது சதவீதம் என்பதை வட்டத் துண்டுகளாகக் காட்டுகிறது.
                                labels, autopct, explode, colors
                            

                            

                                 பரவல் வரைபடம் (Scatter Plot)
                                plt.scatter(x, y)
                                இரண்டு மாறிகளுக்கு இடையிலான உறவை (உறவுமுறை / Correlation) புள்ளிகளாகக் காட்டுகிறது.
                                color, marker, s (size)
                            

                            

                                 நிகழ்வெண் செவ்வகம் (Histogram)
                                plt.hist(data, bins=...)
                                தொடர்ச்சியான எண் தரவுகளின் பரவலை (Frequency Distribution) குறிப்பிட்ட இடைவெளிகளில் காட்டுகிறது.
                                bins, edgecolor, color
                            

                        
                    

                


                
                

                     முக்கிய மேட்பிளாட்லிப் செயற்கூறுகள் (Key Functions)
                

                

                    

                        plt.title("...")
                        வரைபடத்திற்கு மேலே முதன்மை தலைப்பை அமைக்கும்.
                    

                    

                        plt.xlabel("...")
                        X-அச்சுக்கு விளக்க லேபிளை அமைக்கும்.
                    

                    

                        plt.ylabel("...")
                        Y-அச்சுக்கு விளக்க லேபிளை அமைக்கும்.
                    

                    

                        plt.show()
                        உருவாக்கிய வரைபடத்தை புதிய சாளரத்தில் திரையிடும்.
                    

                    

                        plt.grid(True)
                        மதிப்புகளைத் துல்லியமாக அறிய பின்னணியில் கட்டங்களை காட்டும்.
                    

                    

                        plt.legend()
                        வரைபடத்தில் உள்ள குறியீடுகளின் விளக்கப்பெட்டியை சேர்க்கும்.
                    

                    

                        plt.savefig('file.png')
                        வரைபடத்தை படக் கோப்பாக கணினியில் சேமிக்கும்.
                    

                    

                        autopct='%1.1f%%'
                        வட்ட வரைபடத்தில் சதவீத மதிப்பை தசமப் புள்ளியுடன் காட்டும்.

                    செய்முறைப் பயிற்சி (Lab Practical)
                    
12-ஆம் வகுப்பு கணிணி அறிவியல் மாதிரி நிரல்கள்

                    
பாடப்புத்தகத்தில் உள்ள முக்கியமான செய்முறை வினாக்களுக்கான முழுமையான பைதான் குறியீடுகள்.

                


                


                    
                    

                        

                            நிரல் 1: கோட்டு வரைபடம் (Line Chart)
                            
                        

                        
import matplotlib.pyplot as plt

# ஆண்டுகளும் விற்பனையும்
years = [2018, 2019, 2020, 2021, 2022]
sales = [45, 60, 55, 80, 95]

# கோட்டு வரைபடம் வரைதல்
plt.plot(years, sales, color='blue', marker='o', linewidth=2)

# தலைப்பு மற்றும் அச்சு பெயர்கள்
plt.title("வருடாந்திர விற்பனை வளர்ச்சி")
plt.xlabel("ஆண்டுகள்")
plt.ylabel("விற்பனை (ரூபாய் லட்சங்களில்)")

# கட்டக் கோடுகளை இயக்குதல்
plt.grid(True)

# வரைபடத்தை காட்டுதல்
plt.show()

                    


                    
                    

                        

                            நிரல் 2: பட்டை வரைபடம் (Bar Chart)
                            
                        

                        
import matplotlib.pyplot as plt

# பாடங்களும் சராசரி மதிப்பெண்களும்
subjects = ['தமிழ்', 'ஆங்கிலம்', 'கணிதம்', 'அறிவியல்', 'கணிணி']
marks = [85, 78, 92, 88, 98]

# பட்டை வரைபடம் உருவாக்குதல்
plt.bar(subjects, marks, color=['red', 'blue', 'green', 'orange', 'purple'])

# தலைப்பு மற்றும் அச்சு பெயர்கள்
plt.title("வகுப்பு சராசரி மதிப்பெண்கள்")
plt.xlabel("பாடங்கள்")
plt.ylabel("மதிப்பெண்கள் (100-க்கு)")

# வரைபடத்தை காட்டுதல்
plt.show()

                    


                    
                    

                        

                            நிரல் 3: வட்ட வரைபடம் (Pie Chart with explode)
                            
                        

                        
import matplotlib.pyplot as plt

# பொழுதுபோக்குகளின் நேரம் (சதவீதம்)
activities = ['படிப்பு', 'விளையாட்டு', 'தூக்கம்', 'பிற']
hours = [6, 2, 8, 4]

# முதல் துண்டை தனியாகப் பிரித்துக் காட்ட
my_explode = [0.1, 0, 0, 0]

# வட்ட வரைபடம் வரைதல்
plt.pie(hours, labels=activities, explode=my_explode, 
        autopct='%1.1f%%', shadow=True)

plt.title("மாணவரின் தினசரி நேரப் பகிர்வு")
plt.show()

                    


                    
                    

                        

                            நிரல் 4: பரவல் வரைபடம் (Scatter Plot)
                            
                        

                        
import matplotlib.pyplot as plt

# உயரம் (செ.மீ) மற்றும் எடை (கிலோ)
height = [140, 145, 150, 155, 160, 165, 170]
weight = [45, 48, 52, 56, 61, 65, 70]

# பரவல் வரைபடம் வரைதல்
plt.scatter(height, weight, color='red', marker='o', s=60)

plt.title("உயரம் மற்றும் எடை தொடர்பு")
plt.xlabel("உயரம் (செ.மீ)")
plt.ylabel("எடை (கிலோ)")
plt.grid(True)

plt.show()
